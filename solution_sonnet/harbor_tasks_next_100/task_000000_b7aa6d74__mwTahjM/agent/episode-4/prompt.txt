async def collect_all_metrics():
    """Collect metrics from all services concurrently."""
    async with aiohttp.ClientSession() as session:
        tasks = [
            fetch_service_metrics(session, service_name, url)
            for service_name, url in SERVICES.items()
        ]
        results = await asyncio.gather(*tasks)

        # Map results back to service names
        return dict(zip(SERVICES.keys(), results))


def generate_report():
    """Generate the compliance audit report."""
    try:
        metrics = asyncio.run(collect_all_metrics())
    except Exception as e:
        print(f"Collection failed: {e}")
        metrics = {"auth": None, "billing": None, "inventory": None}

    report = {
        "timestamp": datetime.utcnow().isoformat() + "Z",
        "auth": metrics.get("auth"),
        "billing": metrics.get("billing"),
        "inventory": metrics.get("inventory"),
    }

    with open("/home/user/audit/report.json", "w") as f:
        json.dump(report, f, indent=2)

    return report


if __name__ == "__main__":
    report = generate_report()
    print(json.dumps(report, indent=2))
