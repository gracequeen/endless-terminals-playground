def get_latest_tag():
    """Get the latest git tag."""
    cmd = "git describe --tags --abbrev=0"
    result = subprocess.run(cmd, shell=True, capture_output=True, text=True)
    if result.returncode != 0:
        return None
    tag = result.stdout.strip()
    # Remove 'v' prefix if present
    if tag.startswith('v'):
        tag = tag[1:]
    return tag
