# Data pipeline Makefile
# Cleans CSV files from raw/ and aggregates them into final/summary.csv

# Source CSV files in raw/
RAW_CSVS := $(wildcard raw/*.csv)

# Target cleaned CSV files in cleaned/
CLEANED_CSVS := $(patsubst raw/%.csv,cleaned/%.csv,$(RAW_CSVS))

# Default target
all: final/summary.csv

# Rule to clean individual CSV files
cleaned/%.csv: raw/%.csv clean.py
@mkdir -p cleaned
python3 clean.py $< $@
@echo "Cleaned $< -> $@"

# Rule to aggregate all cleaned files
final/summary.csv: $(CLEANED_CSVS) aggregate.py
@mkdir -p final
python3 aggregate.py

# Clean up generated files
clean:
rm -f cleaned/*.csv final/summary.csv
@echo "Cleaned up generated files"

.PHONY: all clean
