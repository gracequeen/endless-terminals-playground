def check_connection(src_ip, dest_port, rules, default_action):
    """Check if a connection is allowed by firewall rules.
    Returns (allowed, matching_rule) tuple.
    """
    ip_obj = ipaddress.ip_address(src_ip)

    # Check rules in order (most specific first)
    for rule in rules:
        if ip_obj in rule['network']:
            # IP matches this CIDR, check port
            if port_in_range(dest_port, rule['port_range']):
                # This rule applies
                return (rule['action'] == 'allow', rule)

    # No matching rule, use default action
    return (default_action == 'allow', None)
