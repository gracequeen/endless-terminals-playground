DROP TABLE IF EXISTS responses_clean;
CREATE TABLE responses_clean AS
-- Keep earliest submission for each non-NULL email
SELECT r.*
FROM responses r
INNER JOIN (
    SELECT email, MIN(submitted_at) as first_sub
    FROM responses
    WHERE email IS NOT NULL AND email != ''
    GROUP BY email
) earliest ON r.email = earliest.email AND r.submitted_at = earliest.first_sub
UNION ALL
-- Keep all rows with NULL/empty email (can't deduplicate these)
SELECT *
FROM responses
WHERE email IS NULL OR email = '';
