DROP TABLE IF EXISTS responses_clean;
CREATE TABLE responses_clean AS
SELECT r.*
FROM responses r
INNER JOIN (
    SELECT email, MIN(submitted_at) as first_sub
    FROM responses
    WHERE email IS NOT NULL
    GROUP BY email
) earliest ON r.email = earliest.email AND r.submitted_at = earliest.first_sub
UNION ALL
SELECT r.*
FROM responses r
WHERE r.email IS NULL
AND r.submitted_at = (
    SELECT MIN(submitted_at)
    FROM responses
    WHERE email IS NULL
);
