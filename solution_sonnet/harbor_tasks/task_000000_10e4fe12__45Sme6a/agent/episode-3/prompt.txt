#!/bin/bash

# Backup script that copies readable files from /home/user/projects to /home/user/backups
# and creates a manifest with ownership information

SOURCE_DIR="/home/user/projects"
BACKUP_DIR="/home/user/backups"
MANIFEST="${BACKUP_DIR}/manifest.txt"

# Initialize manifest
echo "Backup Manifest - $(date)" > "$MANIFEST"
echo "Format: filepath owner:group" >> "$MANIFEST"
echo "" >> "$MANIFEST"

# Find all files in source directory (suppress permission errors)
find "$SOURCE_DIR" -type f 2>/dev/null | while read -r filepath; do
    # Test if file is readable
    if [ -r "$filepath" ]; then
        # Get relative path from source directory
        relpath="${filepath#$SOURCE_DIR/}"

        # Get ownership information
        ownership=$(stat -c '%U:%G' "$filepath" 2>/dev/null)

        # Create target directory structure
        target_file="${BACKUP_DIR}/${relpath}"
        target_dir=$(dirname "$target_file")
        mkdir -p "$target_dir" 2>/dev/null

        # Copy the file
        if cp "$filepath" "$target_file" 2>/dev/null; then
            # Log to manifest
            echo "${relpath} ${ownership}" >> "$MANIFEST"
        fi
    fi
done

echo "Backup complete. See ${MANIFEST} for details."
