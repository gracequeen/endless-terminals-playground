"""
Helper functions for parsing transaction data.
"""
import re


def parse_amount(s):
    """
    Parse an amount string like "$1,234.56" or "€1,234.56" and return a float.
    Strips currency symbols and thousand separators.
    """
    # Remove currency symbols (dollar, euro, etc.) and commas
    cleaned = re.sub(r'[\$€£¥,]', '', s)
    return float(cleaned)
